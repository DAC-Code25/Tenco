cmake_minimum_required(VERSION 3.16)
project(tenco_oak_camera LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
find_package(Threads REQUIRED)
find_package(depthai CONFIG REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc videoio)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil)
add_library(oak_camera_core STATIC src/oak_camera_device.cpp src/oak_native_recorder.cpp)
target_include_directories(oak_camera_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(oak_camera_core PUBLIC Threads::Threads ${OpenCV_LIBS} PkgConfig::AV)
target_compile_options(oak_camera_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(oak_depthai_source STATIC src/depthai_capture_source.cpp)
target_link_libraries(oak_depthai_source PUBLIC oak_camera_core depthai::core)
add_executable(oak_camera_service src/oak_camera_service_main.cpp)
target_link_libraries(oak_camera_service PRIVATE oak_depthai_source)
target_compile_options(oak_camera_service PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_program(FFPROBE_EXECUTABLE ffprobe REQUIRED)
    find_program(FFMPEG_EXECUTABLE ffmpeg REQUIRED)
    add_executable(oak_native_recorder_test tests/test_native_recorder.cpp)
    target_link_libraries(oak_native_recorder_test PRIVATE oak_camera_core)
    add_test(NAME oak_native_recording COMMAND oak_native_recorder_test)
    set_tests_properties(oak_native_recording PROPERTIES TIMEOUT 30)
    add_executable(oak_recovery_test tests/test_recovery.cpp tests/fake_capture_source.cpp)
    target_link_libraries(oak_recovery_test PRIVATE oak_camera_core)
    add_test(NAME oak_capture_recovery COMMAND oak_recovery_test)
    set_tests_properties(oak_capture_recovery PROPERTIES TIMEOUT 30)
    add_executable(oak_camera_service_fixture src/oak_camera_service_main.cpp tests/fake_capture_source.cpp)
    target_link_libraries(oak_camera_service_fixture PRIVATE oak_camera_core)
    add_test(NAME oak_http_recovery COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_http.py $<TARGET_FILE:oak_camera_service_fixture>)
    set_tests_properties(oak_http_recovery PROPERTIES TIMEOUT 30)
endif()
